# ---------- Rule 1 of 1: Protect sensitive files and WordPress  (action: Block) ----------
(http.request.uri.path contains "/." and not starts_with(http.request.uri.path, "/.well-known/"))
or (http.request.uri.path.extension in {"bak" "conf" "config" "dist" "inc" "ini" "log" "orig" "psd" "save" "sh" "sql" "swo" "swp"})
or (ends_with(http.request.uri.path, "/composer.json") or ends_with(http.request.uri.path, "/composer.lock") or ends_with(http.request.uri.path, "/package.json") or ends_with(http.request.uri.path, "/package-lock.json") or ends_with(http.request.uri.path, "/yarn.lock"))
or (http.request.uri.path eq "/xmlrpc.php")
or (http.request.uri.path eq "/wp-config.php")
or (starts_with(http.request.uri.path, "/wp-content/uploads/") and http.request.uri.path.extension in {"php" "php5" "php7" "php8" "phtml" "phar"})
or ((starts_with(http.request.uri.query, "author=") or http.request.uri.query contains "&author=") and not starts_with(http.request.uri.path, "/wp-admin"))
